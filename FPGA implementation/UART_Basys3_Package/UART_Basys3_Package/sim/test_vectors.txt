//   0xFF = 1111 1111 — all HIGH data bits (max stress)
//   0x00 = 0000 0000 — all LOW data bits (zero edge case)
//   0x0F = 0000 1111 — lower nibble HIGH, upper nibble LOW
//   0xF0 = 1111 0000 — upper nibble HIGH, lower nibble LOW
